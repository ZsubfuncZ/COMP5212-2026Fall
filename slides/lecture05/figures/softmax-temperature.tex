% Analytical probabilities for fixed logits (2,1,0), using subtract-max expressions.
% All finite T>0 have the same argmax. Original plot; no empirical calibration claim.
\begin{tikzpicture}
\begin{axis}[
  width=\linewidth,height=4.2cm,
  xmin=0.25,xmax=4,ymin=0,ymax=1,
  xlabel={Temperature $T$},ylabel={Probability},
  xtick={0.5,1,2,3,4},ytick={0,0.333333,0.666667,1},
  yticklabels={$0$,$1/3$,$2/3$,$1$},
  font=\scriptsize,tick label style={font=\scriptsize},
  axis line style={black!45},tick style={black!45},
  grid=major,grid style={black!10},scaled ticks=false,
  legend style={font=\scriptsize,draw=none,fill=white,
    at={(0.98,0.98)},anchor=north east,inner sep=2pt},
  domain=0.25:4,samples=100
]
\addplot[blueplot,thick,no marks] {1/(1+exp(-1/x)+exp(-2/x))};
\addlegendentry{$z_1=2$}
\addplot[redplot,thick,no marks] {exp(-1/x)/(1+exp(-1/x)+exp(-2/x))};
\addlegendentry{$z_2=1$}
\addplot[bulletgreen!60!black,thick,no marks] {exp(-2/x)/(1+exp(-1/x)+exp(-2/x))};
\addlegendentry{$z_3=0$}
\addplot[black!40,dashed,no marks,forget plot] coordinates {(0.25,0.333333) (4,0.333333)};
\addplot[black!35,densely dotted,no marks,forget plot] coordinates {(1,0) (1,1)};
\end{axis}
\end{tikzpicture}
